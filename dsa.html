int solve(int[] a){
  int l = 0, r = 1_000_000_000;
  while(l < r){
    int m = l + (r - l)/2;
    if(check(a, m)) r = m; else l = m + 1;
  }
  return l;
}
// boolean check(...) implements feasibility
Sliding Window (max length with constraint) 
int atMostKDistinct(int[] a, int k){
  Map<Integer,Integer> cnt = new HashMap<>();
  int i=0, best=0;
  for(int j=0;j<a.length;j++){
    cnt.merge(a[j],1,Integer::sum);
    while(cnt.size() > k){
      cnt.put(a[i], cnt.get(a[i])-1);
      if(cnt.get(a[i])==0) cnt.remove(a[i]);
      i++;
    }
    best = Math.max(best, j - i + 1);
  }
  return best;
}
Graph: BFS & Dijkstra 
int bfs(int n, List<List<Integer>> g, int s){
  Queue<Integer> q = new ArrayDeque<>();
  boolean[] seen = new boolean[n];
  int[] dist = new int[n];
  q.add(s); seen[s]=true;
  while(!q.isEmpty()){
    int u=q.poll();
    for(int v:g.get(u)){
      if(!seen[v]){seen[v]=true;dist[v]=dist[u]+1;q.add(v);}
    }
  }
  return Arrays.stream(dist).max().getAsInt();
}

Common Pitfalls

  • Off-by-one in sliding windows and binary search bounds.
  • Forgetting to mark visited or to check edge cases (empty, single-element).
  • Inefficient data structures (e.g., LinkedList random access).
